## Reference implementation

Open only after making an independent attempt.

```rust
pub fn split_label<'text>(
    text: &'text str,
    separator: &str,
) -> Option<(&'text str, &'text str)> {
    if separator.is_empty() {
        return None;
    }

    let start = text.find(separator)?;
    let end = start.checked_add(separator.len())?;
    Some((text.get(..start)?, text.get(end..)?))
}

fn main() {
    let mut label = String::from("部署：完成");
    let (kind, value) = split_label(&label, "：").unwrap();
    println!("parts: {kind} / {value}");
    label.push('!');
    println!("updated: {label}");
}
```
